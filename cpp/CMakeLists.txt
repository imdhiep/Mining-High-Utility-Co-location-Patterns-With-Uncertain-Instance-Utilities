cmake_minimum_required(VERSION 3.16)
project(colocation_mining CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O2")
add_compile_options(-Wall -Wextra)

file(GLOB_RECURSE LIBRARY_SOURCES CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/src/*.cpp)
add_library(colocation STATIC ${LIBRARY_SOURCES})
target_include_directories(colocation PUBLIC ${CMAKE_SOURCE_DIR}/src)

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR}/bin)
foreach(app uhucpm_mine pso_mucp post_counts fsq_prep yelp_prep yelp_reviews pattern_scores)
  add_executable(${app} apps/${app}.cpp)
  target_link_libraries(${app} PRIVATE colocation)
endforeach()

file(GLOB UNIT_TESTS CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/tests/unit/*_test.cpp)
if(UNIT_TESTS)
  add_executable(unit_tests ${UNIT_TESTS} tests/unit/test_main.cpp)
  target_link_libraries(unit_tests PRIVATE colocation)
  target_include_directories(unit_tests PRIVATE ${CMAKE_SOURCE_DIR}/tests/unit)
  enable_testing()
  add_test(NAME unit_tests COMMAND unit_tests)
endif()
